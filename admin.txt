        function adminChangeVolume(cat, val, labelId) {
            GornitsaMedia.setVolume(cat, val);
            const el = document.getElementById(labelId);
            if (el) el.textContent = Math.round(val) + '%';
        }

        async function adminUploadLocalFile(cat) {
            const fileInp = document.getElementById('fileInput_' + cat);
            const titleInp = document.getElementById('fileTitle_' + cat);
            const file = fileInp && fileInp.files && fileInp.files[0];
            if (!file) {
                alert('Сначала выберите аудио- или видеофайл на компьютере!');
                return;
            }
            const customTitle = titleInp ? titleInp.value.trim() : '';
            await GornitsaMedia.addUploadedFile(cat, file, customTitle);
            if (fileInp) fileInp.value = '';
